// ============================================================================
// File Name   : SingleCycle_RISCV_AllInOne.v
// Description : Trọn bộ vi xử lý RISC-V Single-Cycle (Datapath + Control + TB)
// ISA Subset  : ADD, SUB, AND, OR, SLT, ADDI, LW, SW, BEQ
// ============================================================================

`timescale 1ns/1ps

// ============================================================================
// Module: Control_Unit
// ============================================================================
module Control_Unit (
    input  wire [6:0] op,
    input  wire [2:0] funct3,
    input  wire       funct7_5,
    output reg        RegWrite,
    output reg        ALUSrc,
    output reg        MemWrite,
    output reg  [2:0] ImmSrc,
    output reg  [2:0] ResultSrc,
    output reg  [3:0] ALUControl,
    output reg        Branch
);
    localparam IMM_I = 3'b000;
    localparam IMM_S = 3'b001;
    localparam IMM_B = 3'b010;

    localparam RES_ALU = 3'b000;
    localparam RES_MEM = 3'b001;
    localparam RES_PC4 = 3'b010;

    localparam ALU_ADD = 4'b0000;
    localparam ALU_SUB = 4'b0001;
    localparam ALU_AND = 4'b0010;
    localparam ALU_OR  = 4'b0011;
    localparam ALU_SLT = 4'b0101;

    always @(*) begin
        RegWrite   = 1'b0;
        ALUSrc     = 1'b0;
        MemWrite   = 1'b0;
        ImmSrc     = IMM_I;
        ResultSrc  = RES_ALU;
        ALUControl = ALU_ADD;
        Branch     = 1'b0;

        case (op)
            7'b0000011: begin // LW
                RegWrite   = 1'b1;
                ALUSrc     = 1'b1;
                ImmSrc     = IMM_I;
                ResultSrc  = RES_MEM;
                ALUControl = ALU_ADD;
            end

            7'b0100011: begin // SW
                ALUSrc     = 1'b1;
                MemWrite   = 1'b1;
                ImmSrc     = IMM_S;
                ALUControl = ALU_ADD;
            end

            7'b0110011: begin // R-type: ADD, SUB, AND, OR, SLT
                RegWrite = 1'b1;
                case (funct3)
                    3'b000: ALUControl = funct7_5 ? ALU_SUB : ALU_ADD;
                    3'b111: ALUControl = ALU_AND;
                    3'b110: ALUControl = ALU_OR;
                    3'b010: ALUControl = ALU_SLT;
                    default: begin RegWrite = 1'b0; ALUControl = ALU_ADD; end
                endcase
            end

            7'b0010011: begin // ADDI
                RegWrite   = (funct3 == 3'b000);
                ALUSrc     = 1'b1;
                ImmSrc     = IMM_I;
                ALUControl = ALU_ADD;
            end

            7'b1100011: begin // BEQ
                Branch     = (funct3 == 3'b000);
                ImmSrc     = IMM_B;
                ALUControl = ALU_SUB;
            end

            default: begin
                // Trạng thái mặc định an toàn cho NOP / Bubble
            end
        endcase
    end
endmodule

// ============================================================================
// Module: Extend (Sign Extension Unit)
// ============================================================================
module Extend (
    input  wire [2:0]  ImmSrc,
    input  wire [24:0] Imm,
    output reg  [31:0] ImmExt
);
    always @(*) begin
        case (ImmSrc)
            3'b000: ImmExt = {{20{Imm[24]}}, Imm[24:13]};
            3'b001: ImmExt = {{20{Imm[24]}}, Imm[24:18], Imm[4:0]};
            3'b010: ImmExt = {{19{Imm[24]}}, Imm[24], Imm[0], Imm[23:18], Imm[4:1], 1'b0};
            default: ImmExt = 32'd0;
        endcase
    end
endmodule

// ============================================================================
// Module: alu
// ============================================================================
module alu (
    input  wire [3:0]  alu_op,
    input  wire [31:0] src_a,
    input  wire [31:0] src_b,
    output reg  [31:0] alu_result,
    output wire        Zero
);
    localparam ALU_ADD = 4'b0000;
    localparam ALU_SUB = 4'b0001;
    localparam ALU_AND = 4'b0010;
    localparam ALU_OR  = 4'b0011;
    localparam ALU_SLT = 4'b0101;

    assign Zero = (alu_result == 32'd0);

    always @(*) begin
        case (alu_op)
            ALU_ADD: alu_result = src_a + src_b;
            ALU_SUB: alu_result = src_a - src_b;
            ALU_AND: alu_result = src_a & src_b;
            ALU_OR : alu_result = src_a | src_b;
            ALU_SLT: alu_result = ($signed(src_a) < $signed(src_b)) ? 32'd1 : 32'd0;
            default: alu_result = 32'd0;
        endcase
    end
endmodule

// ============================================================================
// Module: Register_File
// ============================================================================
module Register_File (
    input  wire        clk,
    input  wire        RegWrite,
    input  wire [4:0]  read_addr_1,
    input  wire [4:0]  read_addr_2,
    input  wire [4:0]  write_addr,
    input  wire [31:0] write_data,
    output wire [31:0] read_data_1,
    output wire [31:0] read_data_2
);
    reg [31:0] Regfile [0:31];
    integer k;

    initial begin
        for (k = 0; k < 32; k = k + 1)
            Regfile[k] = 32'd0;
    end

    // x0 luôn cố định giá trị 0
    assign read_data_1 = (read_addr_1 == 5'd0) ? 32'd0 : Regfile[read_addr_1];
    assign read_data_2 = (read_addr_2 == 5'd0) ? 32'd0 : Regfile[read_addr_2];

    always @(posedge clk) begin
        if (RegWrite && (write_addr != 5'd0))
            Regfile[write_addr] <= write_data;
    end
endmodule

// ============================================================================
// Module: Instruction_Memory (ROM)
// ============================================================================
module Instruction_Memory (
    input  wire [31:0] read_address,
    output wire [31:0] instruction
);
    reg [31:0] Imemory [0:63];
    integer i;

    assign instruction = (read_address[31:8] == 24'd0)
                       ? Imemory[read_address[31:2]]
                       : 32'h00000013; // NOP (ADDI x0, x0, 0)

    initial begin
        for (i = 0; i < 64; i = i + 1)
            Imemory[i] = 32'h00000013;

        Imemory[0] = 32'h00500093; // 0x00: addi x1, x0, 5
        Imemory[1] = 32'h00a00113; // 0x04: addi x2, x0, 10
        Imemory[2] = 32'h002081b3; // 0x08: add  x3, x1, x2
        Imemory[3] = 32'h00302223; // 0x0C: sw   x3, 4(x0)
        Imemory[4] = 32'h00402203; // 0x10: lw   x4, 4(x0)
        Imemory[5] = 32'h401202b3; // 0x14: sub  x5, x4, x1
        Imemory[6] = 32'h00228463; // 0x18: beq  x5, x2, +8
        Imemory[7] = 32'h06300313; // 0x1C: addi x6, x0, 99 (Branch nhảy qua lệnh này)
        Imemory[8] = 32'h00128393; // 0x20: addi x7, x5, 1
    end
endmodule

// ============================================================================
// Module: Data_Memory (RAM)
// ============================================================================
module Data_Memory (
    input  wire        clk,
    input  wire        MemWrite,
    input  wire [31:0] addr,
    input  wire [31:0] write_data,
    output wire [31:0] read_data
);
    reg [31:0] RAM [0:63];
    integer i;
    wire valid_addr = (addr[31:8] == 24'd0);

    initial begin
        for (i = 0; i < 64; i = i + 1)
            RAM[i] = 32'd0;
    end

    // Đọc bất đồng bộ trong chu kỳ Single-Cycle
    assign read_data = valid_addr ? RAM[addr[31:2]] : 32'd0;

    always @(posedge clk) begin
        if (MemWrite && valid_addr)
            RAM[addr[31:2]] <= write_data;
    end
endmodule

// ============================================================================
// Module: Program_Counter
// ============================================================================
module Program_Counter (
    input  wire        clk,
    input  wire        reset_n,
    input  wire [31:0] d,
    output reg  [31:0] q
);
    always @(posedge clk or negedge reset_n) begin
        if (!reset_n)
            q <= 32'd0;
        else
            q <= d;
    end
endmodule

// ============================================================================
// Module: SingleCycle_RISCV (Top Datapath)
// ============================================================================
module SingleCycle_RISCV (
    input  wire        clk,
    input  wire        reset_n,
    output wire [31:0] PC,
    output wire [31:0] Instr,
    output wire [31:0] ALUResult,
    output wire [31:0] WriteData,
    output wire [31:0] Result
);
    // Instruction field breakdown
    wire [6:0] op       = Instr[6:0];
    wire [2:0] funct3   = Instr[14:12];
    wire       funct7_5 = Instr[30];
    wire [4:0] rs1      = Instr[19:15];
    wire [4:0] rs2      = Instr[24:20];
    wire [4:0] rd       = Instr[11:7];

    // Control signals
    wire       RegWrite;
    wire       ALUSrc;
    wire       MemWrite;
    wire [2:0] ImmSrc;
    wire [2:0] ResultSrc;
    wire [3:0] ALUControl;
    wire       Branch;

    // Datapath signals
    wire [31:0] PCNext, PCPlus4, PCTarget;
    wire [31:0] RD1, RD2, ImmExt, SrcA, SrcB;
    wire [31:0] ReadData;
    wire        Zero, PCSrc;

    // --- FETCH ---
    assign PCPlus4 = PC + 32'd4;
    assign PCNext  = PCSrc ? PCTarget : PCPlus4;

    Program_Counter PC_Reg (
        .clk(clk),
        .reset_n(reset_n),
        .d(PCNext),
        .q(PC)
    );

    Instruction_Memory IMEM (
        .read_address(PC),
        .instruction(Instr)
    );

    // --- DECODE & CONTROL ---
    Control_Unit Control (
        .op(op),
        .funct3(funct3),
        .funct7_5(funct7_5),
        .RegWrite(RegWrite),
        .ALUSrc(ALUSrc),
        .MemWrite(MemWrite),
        .ImmSrc(ImmSrc),
        .ResultSrc(ResultSrc),
        .ALUControl(ALUControl),
        .Branch(Branch)
    );

    Register_File RegFiles (
        .clk(clk),
        .RegWrite(RegWrite),
        .read_addr_1(rs1),
        .read_addr_2(rs2),
        .write_addr(rd),
        .write_data(Result),
        .read_data_1(RD1),
        .read_data_2(RD2)
    );

    Extend Imm_Ext (
        .ImmSrc(ImmSrc),
        .Imm(Instr[31:7]),
        .ImmExt(ImmExt)
    );

    // --- EXECUTE ---
    assign SrcA     = RD1;
    assign SrcB     = ALUSrc ? ImmExt : RD2;
    assign WriteData = RD2;

    alu Core_ALU (
        .alu_op(ALUControl),
        .src_a(SrcA),
        .src_b(SrcB),
        .alu_result(ALUResult),
        .Zero(Zero)
    );

    assign PCTarget = PC + ImmExt;
    assign PCSrc    = Branch && Zero;

    // --- MEMORY ---
    Data_Memory DMEM (
        .clk(clk),
        .MemWrite(MemWrite),
        .addr(ALUResult),
        .write_data(WriteData),
        .read_data(ReadData)
    );

    // --- WRITEBACK ---
    assign Result = (ResultSrc == 3'b001) ? ReadData :
                    (ResultSrc == 3'b010) ? PCPlus4  : ALUResult;

endmodule

// ============================================================================
// Module: SingleCycle_RISCV_tb (Self-Checking Testbench)
// ============================================================================
module SingleCycle_RISCV_tb;
    reg         clk;
    reg         reset_n;

    wire [31:0] PC;
    wire [31:0] Instr;
    wire [31:0] ALUResult;
    wire [31:0] WriteData;
    wire [31:0] Result;

    SingleCycle_RISCV uut (
        .clk(clk),
        .reset_n(reset_n),
        .PC(PC),
        .Instr(Instr),
        .ALUResult(ALUResult),
        .WriteData(WriteData),
        .Result(Result)
    );

    // 100MHz simulation clock (Period = 10ns)
    always #5 clk = ~clk;

    initial begin
        clk     = 1'b0;
        reset_n = 1'b0;
        $dumpfile("SingleCycle_RISCV_tb.vcd");
        $dumpvars(0, SingleCycle_RISCV_tb);

        // Giữ reset 12ns sau đó nhả reset
        #12 reset_n = 1'b1;

        // Chạy đủ 10 chu kỳ xung nhịp để thực thi toàn bộ luồng lệnh
        #100;

        // Đối chiếu giá trị thanh ghi và RAM
        if ((uut.RegFiles.Regfile[1] == 32'd5)  &&
            (uut.RegFiles.Regfile[2] == 32'd10) &&
            (uut.RegFiles.Regfile[3] == 32'd15) &&
            (uut.DMEM.RAM[1]         == 32'd15) &&
            (uut.RegFiles.Regfile[4] == 32'd15) &&
            (uut.RegFiles.Regfile[5] == 32'd10) &&
            (uut.RegFiles.Regfile[6] == 32'd0)  &&
            (uut.RegFiles.Regfile[7] == 32'd11)) begin
            $display("==========================================================");
            $display("PASS: All single-cycle execution and branch tests passed.");
            $display("==========================================================");
        end else begin
            $display("==========================================================");
            $display("FAIL: Data mismatch detected!");
            $display("x1=%0d x2=%0d x3=%0d mem[1]=%0d x4=%0d x5=%0d x6=%0d x7=%0d",
                     uut.RegFiles.Regfile[1], uut.RegFiles.Regfile[2],
                     uut.RegFiles.Regfile[3], uut.DMEM.RAM[1],
                     uut.RegFiles.Regfile[4], uut.RegFiles.Regfile[5],
                     uut.RegFiles.Regfile[6], uut.RegFiles.Regfile[7]);
            $display("==========================================================");
        end
        $finish;
    end
endmodule
