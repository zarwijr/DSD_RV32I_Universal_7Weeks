// ============================================================================
// File Name   : MultiCycle_RISCV_AllInOne.v
// Description : Trọn bộ vi xử lý RISC-V Multi-Cycle (FSM Controller + Datapath + TB)
// ISA Subset  : ADD, SUB, AND, OR, SLT, ADDI, LW, SW, BEQ
// ============================================================================

`timescale 1ns/1ps

// ============================================================================
// Module: MultiCycle_Control_Unit (FSM Main Controller)
// ============================================================================
module MultiCycle_Control_Unit (
    input  wire       clk,
    input  wire       reset_n,
    input  wire [6:0] op,
    input  wire [2:0] funct3,
    input  wire       funct7_5,
    input  wire       Zero,
    output reg        PCWrite,
    output reg        IRWrite,
    output reg        RegWrite,
    output reg        MemWrite,
    output reg  [1:0] ALUSrcA,
    output reg  [1:0] ALUSrcB,
    output reg  [1:0] ResultSrc,
    output reg  [3:0] ALUControl,
    output reg  [2:0] ImmSrc,
    output reg  [3:0] state_out
);
    // FSM States Definition
    localparam S_FETCH   = 4'd0;
    localparam S_DECODE  = 4'd1;
    localparam S_MEM_ADR = 4'd2;
    localparam S_MEM_RD  = 4'd3;
    localparam S_MEM_WB  = 4'd4;
    localparam S_MEM_WR  = 4'd5;
    localparam S_EXEC_R  = 4'd6;
    localparam S_ALU_WB  = 4'd7;
    localparam S_EXEC_I  = 4'd8;
    localparam S_BRANCH  = 4'd9;

    // ImmSrc constants
    localparam IMM_I = 3'b000;
    localparam IMM_S = 3'b001;
    localparam IMM_B = 3'b010;

    // ALU Operations
    localparam ALU_ADD = 4'b0000;
    localparam ALU_SUB = 4'b0001;
    localparam ALU_AND = 4'b0010;
    localparam ALU_OR  = 4'b0011;
    localparam ALU_SLT = 4'b0101;

    reg [3:0] state, next_state;

    // State Register
    always @(posedge clk or negedge reset_n) begin
        if (!reset_n)
            state <= S_FETCH;
        else
            state <= next_state;
    end

    // Next-State Logic
    always @(*) begin
        case (state)
            S_FETCH:   next_state = S_DECODE;
            
            S_DECODE: begin
                case (op)
                    7'b0000011: next_state = S_MEM_ADR; // LW
                    7'b0100011: next_state = S_MEM_ADR; // SW
                    7'b0110011: next_state = S_EXEC_R;  // R-type
                    7'b0010011: next_state = S_EXEC_I;  // ADDI
                    7'b1100011: next_state = S_BRANCH;  // BEQ
                    default:    next_state = S_FETCH;
                endcase
            end

            S_MEM_ADR: begin
                if (op == 7'b0000011)
                    next_state = S_MEM_RD;
                else
                    next_state = S_MEM_WR;
            end

            S_MEM_RD:  next_state = S_MEM_WB;
            S_MEM_WB:  next_state = S_FETCH;
            S_MEM_WR:  next_state = S_FETCH;
            S_EXEC_R:  next_state = S_ALU_WB;
            S_EXEC_I:  next_state = S_ALU_WB;
            S_ALU_WB:  next_state = S_FETCH;
            S_BRANCH:  next_state = S_FETCH;
            default:   next_state = S_FETCH;
        endcase
    end

    // ImmSrc Decoder
    always @(*) begin
        case (op)
            7'b0000011: ImmSrc = IMM_I;
            7'b0100011: ImmSrc = IMM_S;
            7'b0010011: ImmSrc = IMM_I;
            7'b1100011: ImmSrc = IMM_B;
            default:    ImmSrc = IMM_I;
        endcase
    end

    // Output Control Logic
    always @(*) begin
        PCWrite    = 1'b0;
        IRWrite    = 1'b0;
        RegWrite   = 1'b0;
        MemWrite   = 1'b0;
        ALUSrcA    = 2'b00;
        ALUSrcB    = 2'b00;
        ResultSrc  = 2'b00;
        ALUControl = ALU_ADD;
        state_out  = state;

        case (state)
            S_FETCH: begin
                // PC <= PC + 4, fetch instruction
                ALUSrcA    = 2'b00;       // PC
                ALUSrcB    = 2'b10;       // +4
                ALUControl = ALU_ADD;
                ResultSrc  = 2'b10;       // ALUResult trực tiếp
                PCWrite    = 1'b1;
                IRWrite    = 1'b1;
            end

            S_DECODE: begin
                // Tính trước branch target: OldPC + ImmExt
                ALUSrcA    = 2'b01;       // OldPC
                ALUSrcB    = 2'b01;       // ImmExt
                ALUControl = ALU_ADD;
            end

            S_MEM_ADR: begin
                // Address = rs1 + ImmExt
                ALUSrcA    = 2'b10;       // A (rs1)
                ALUSrcB    = 2'b01;       // ImmExt
                ALUControl = ALU_ADD;
            end

            S_MEM_RD: begin
                // Giữ nguyên trạng thái để chờ RAM đọc dữ liệu vào Data register
            end

            S_MEM_WB: begin
                // Ghi dữ liệu từ Data register vào Register File
                RegWrite   = 1'b1;
                ResultSrc  = 2'b01;       // Data (MDR)
            end

            S_MEM_WR: begin
                // Ghi B (rs2) vào Data Memory
                MemWrite   = 1'b1;
            end

            S_EXEC_R: begin
                // ALUResult = rs1 op rs2
                ALUSrcA = 2'b10;          // A
                ALUSrcB = 2'b00;          // B
                case (funct3)
                    3'b000: ALUControl = funct7_5 ? ALU_SUB : ALU_ADD;
                    3'b111: ALUControl = ALU_AND;
                    3'b110: ALUControl = ALU_OR;
                    3'b010: ALUControl = ALU_SLT;
                    default: ALUControl = ALU_ADD;
                endcase
            end

            S_EXEC_I: begin
                // ALUResult = rs1 + ImmExt
                ALUSrcA    = 2'b10;       // A
                ALUSrcB    = 2'b01;       // ImmExt
                ALUControl = ALU_ADD;
            end

            S_ALU_WB: begin
                // Ghi kết quả từ ALUOut vào Register File
                RegWrite  = 1'b1;
                ResultSrc = 2'b00;        // ALUOut
            end

            S_BRANCH: begin
                // So sánh rs1 - rs2, nếu bằng nhau thì cập nhật PC = ALUOut
                ALUSrcA    = 2'b10;       // A
                ALUSrcB    = 2'b00;       // B
                ALUControl = ALU_SUB;
                if (Zero) begin
                    ResultSrc = 2'b00;    // Lấy ALUOut (đã tính ở S_DECODE)
                    PCWrite   = 1'b1;
                end
            end

            default: begin
                // Trạng thái giữ nguyên mặc định
            end
        endcase
    end
endmodule

// ============================================================================
// Module: Extend (Sign Extension Unit)
// ============================================================================
module Extend (
    input  wire [2:0]  ImmSrc,
    input  wire [24:0] Imm,
    output reg  [31:0] ImmExt
);
    always @(*) begin
        case (ImmSrc)
            3'b000: ImmExt = {{20{Imm[24]}}, Imm[24:13]};
            3'b001: ImmExt = {{20{Imm[24]}}, Imm[24:18], Imm[4:0]};
            3'b010: ImmExt = {{19{Imm[24]}}, Imm[24], Imm[0], Imm[23:18], Imm[4:1], 1'b0};
            default: ImmExt = 32'd0;
        endcase
    end
endmodule

// ============================================================================
// Module: alu (Shared ALU)
// ============================================================================
module alu (
    input  wire [3:0]  alu_op,
    input  wire [31:0] src_a,
    input  wire [31:0] src_b,
    output reg  [31:0] alu_result,
    output wire        Zero
);
    localparam ALU_ADD = 4'b0000;
    localparam ALU_SUB = 4'b0001;
    localparam ALU_AND = 4'b0010;
    localparam ALU_OR  = 4'b0011;
    localparam ALU_SLT = 4'b0101;

    assign Zero = (alu_result == 32'd0);

    always @(*) begin
        case (alu_op)
            ALU_ADD: alu_result = src_a + src_b;
            ALU_SUB: alu_result = src_a - src_b;
            ALU_AND: alu_result = src_a & src_b;
            ALU_OR : alu_result = src_a | src_b;
            ALU_SLT: alu_result = ($signed(src_a) < $signed(src_b)) ? 32'd1 : 32'd0;
            default: alu_result = 32'd0;
        endcase
    end
endmodule

// ============================================================================
// Module: Register_File
// ============================================================================
module Register_File (
    input  wire        clk,
    input  wire        RegWrite,
    input  wire [4:0]  read_addr_1,
    input  wire [4:0]  read_addr_2,
    input  wire [4:0]  write_addr,
    input  wire [31:0] write_data,
    output wire [31:0] read_data_1,
    output wire [31:0] read_data_2
);
    reg [31:0] Regfile [0:31];
    integer k;

    initial begin
        for (k = 0; k < 32; k = k + 1)
            Regfile[k] = 32'd0;
    end

    assign read_data_1 = (read_addr_1 == 5'd0) ? 32'd0 : Regfile[read_addr_1];
    assign read_data_2 = (read_addr_2 == 5'd0) ? 32'd0 : Regfile[read_addr_2];

    always @(posedge clk) begin
        if (RegWrite && (write_addr != 5'd0))
            Regfile[write_addr] <= write_data;
    end
endmodule

// ============================================================================
// Module: Instruction_Memory
// ============================================================================
module Instruction_Memory (
    input  wire [31:0] read_address,
    output wire [31:0] instruction
);
    reg [31:0] Imemory [0:63];
    integer i;

    assign instruction = (read_address[31:8] == 24'd0)
                       ? Imemory[read_address[31:2]]
                       : 32'h00000013; // NOP (ADDI x0, x0, 0)

    initial begin
        for (i = 0; i < 64; i = i + 1)
            Imemory[i] = 32'h00000013;

        Imemory[0] = 32'h00500093; // 0x00: addi x1, x0, 5
        Imemory[1] = 32'h00a00113; // 0x04: addi x2, x0, 10
        Imemory[2] = 32'h002081b3; // 0x08: add  x3, x1, x2
        Imemory[3] = 32'h00302223; // 0x0C: sw   x3, 4(x0)
        Imemory[4] = 32'h00402203; // 0x10: lw   x4, 4(x0)
        Imemory[5] = 32'h401202b3; // 0x14: sub  x5, x4, x1
        Imemory[6] = 32'h00228463; // 0x18: beq  x5, x2, +8
        Imemory[7] = 32'h06300313; // 0x1C: addi x6, x0, 99 (Bị nhảy qua khi BEQ taken)
        Imemory[8] = 32'h00128393; // 0x20: addi x7, x5, 1
    end
endmodule

// ============================================================================
// Module: Data_Memory
// ============================================================================
module Data_Memory (
    input  wire        clk,
    input  wire        MemWrite,
    input  wire [31:0] addr,
    input  wire [31:0] write_data,
    output wire [31:0] read_data
);
    reg [31:0] RAM [0:63];
    integer i;
    wire valid_addr = (addr[31:8] == 24'd0);

    initial begin
        for (i = 0; i < 64; i = i + 1)
            RAM[i] = 32'd0;
    end

    assign read_data = valid_addr ? RAM[addr[31:2]] : 32'd0;

    always @(posedge clk) begin
        if (MemWrite && valid_addr)
            RAM[addr[31:2]] <= write_data;
    end
endmodule

// ============================================================================
// Module: MultiCycle_RISCV (Top Multi-Cycle Core)
// ============================================================================
module MultiCycle_RISCV (
    input  wire        clk,
    input  wire        reset_n,
    output reg  [31:0] PC,
    output reg  [31:0] Instr,
    output reg  [31:0] ALUOut,
    output reg  [31:0] Data,
    output wire [3:0]  State
);
    // Internal holding registers
    reg  [31:0] OldPC;
    reg  [31:0] A, B;

    // Control wires
    wire       PCWrite;
    wire       IRWrite;
    wire       RegWrite;
    wire       MemWrite;
    wire [1:0] ALUSrcA;
    wire [1:0] ALUSrcB;
    wire [1:0] ResultSrc;
    wire [3:0] ALUControl;
    wire [2:0] ImmSrc;

    // Datapath wires
    wire [31:0] InstrF;
    wire [31:0] RD1, RD2;
    wire [31:0] ImmExt;
    reg  [31:0] SrcA, SrcB;
    wire [31:0] ALUResult;
    wire        Zero;
    reg  [31:0] Result;
    wire [31:0] ReadDataM;

    // ------------------------------------------------------------------------
    // FSM Controller Instance
    // ------------------------------------------------------------------------
    MultiCycle_Control_Unit Control (
        .clk(clk),
        .reset_n(reset_n),
        .op(Instr[6:0]),
        .funct3(Instr[14:12]),
        .funct7_5(Instr[30]),
        .Zero(Zero),
        .PCWrite(PCWrite),
        .IRWrite(IRWrite),
        .RegWrite(RegWrite),
        .MemWrite(MemWrite),
        .ALUSrcA(ALUSrcA),
        .ALUSrcB(ALUSrcB),
        .ResultSrc(ResultSrc),
        .ALUControl(ALUControl),
        .ImmSrc(ImmSrc),
        .state_out(State)
    );

    // ------------------------------------------------------------------------
    // Architectural & Internal Holding Registers
    // ------------------------------------------------------------------------
    always @(posedge clk or negedge reset_n) begin
        if (!reset_n) begin
            PC     <= 32'd0;
            OldPC  <= 32'd0;
            Instr  <= 32'h00000013; // NOP
            Data   <= 32'd0;
            A      <= 32'd0;
            B      <= 32'd0;
            ALUOut <= 32'd0;
        end else begin
            if (PCWrite)
                PC <= Result;

            if (IRWrite) begin
                Instr <= InstrF;
                OldPC <= PC;
            end

            Data   <= ReadDataM;
            A      <= RD1;
            B      <= RD2;
            ALUOut <= ALUResult;
        end
    end

    // ------------------------------------------------------------------------
    // Memory Subsystem
    // ------------------------------------------------------------------------
    Instruction_Memory IMEM (
        .read_address(PC),
        .instruction(InstrF)
    );

    Data_Memory DMEM (
        .clk(clk),
        .MemWrite(MemWrite),
        .addr(ALUOut),
        .write_data(B),
        .read_data(ReadDataM)
    );

    // ------------------------------------------------------------------------
    // Register File & Sign Extension
    // ------------------------------------------------------------------------
    Register_File RegFiles (
        .clk(clk),
        .RegWrite(RegWrite),
        .read_addr_1(Instr[19:15]),
        .read_addr_2(Instr[24:20]),
        .write_addr(Instr[11:7]),
        .write_data(Result),
        .read_data_1(RD1),
        .read_data_2(RD2)
    );

    Extend Ext_Unit (
        .ImmSrc(ImmSrc),
        .Imm(Instr[31:7]),
        .ImmExt(ImmExt)
    );

    // ------------------------------------------------------------------------
    // ALU Input Multiplexers
    // ------------------------------------------------------------------------
    always @(*) begin
        case (ALUSrcA)
            2'b00:   SrcA = PC;
            2'b01:   SrcA = OldPC;
            2'b10:   SrcA = A;
            default: SrcA = A;
        endcase
    end

    always @(*) begin
        case (ALUSrcB)
            2'b00:   SrcB = B;
            2'b01:   SrcB = ImmExt;
            2'b10:   SrcB = 32'd4;
            default: SrcB = B;
        endcase
    end

    alu Core_ALU (
        .alu_op(ALUControl),
        .src_a(SrcA),
        .src_b(SrcB),
        .alu_result(ALUResult),
        .Zero(Zero)
    );

    // ------------------------------------------------------------------------
    // Result Multiplexer
    // ------------------------------------------------------------------------
    always @(*) begin
        case (ResultSrc)
            2'b00:   Result = ALUOut;
            2'b01:   Result = Data;
            2'b10:   Result = ALUResult;
            default: Result = ALUOut;
        endcase
    end

endmodule

// ============================================================================
// Module: MultiCycle_RISCV_tb (Self-Checking Testbench)
// ============================================================================
module MultiCycle_RISCV_tb;
    reg         clk;
    reg         reset_n;

    wire [31:0] PC;
    wire [31:0] Instr;
    wire [31:0] ALUOut;
    wire [31:0] Data;
    wire [3:0]  State;

    MultiCycle_RISCV uut (
        .clk(clk),
        .reset_n(reset_n),
        .PC(PC),
        .Instr(Instr),
        .ALUOut(ALUOut),
        .Data(Data),
        .State(State)
    );

    // 100MHz simulation clock (Chu kỳ 10ns)
    always #5 clk = ~clk;

    initial begin
        clk     = 1'b0;
        reset_n = 1'b0;
        $dumpfile("MultiCycle_RISCV_tb.vcd");
        $dumpvars(0, MultiCycle_RISCV_tb);

        // Reset bất đồng bộ trong 12ns
        #12 reset_n = 1'b1;

        // Chạy 40 chu kỳ xung nhịp (~400ns) để hoàn thành toàn bộ chương trình đa chu kỳ
        #400;

        // Tự động kiểm tra giá trị thanh ghi và bộ nhớ RAM
        if ((uut.RegFiles.Regfile[1] == 32'd5)  &&
            (uut.RegFiles.Regfile[2] == 32'd10) &&
            (uut.RegFiles.Regfile[3] == 32'd15) &&
            (uut.DMEM.RAM[1]         == 32'd15) &&
            (uut.RegFiles.Regfile[4] == 32'd15) &&
            (uut.RegFiles.Regfile[5] == 32'd10) &&
            (uut.RegFiles.Regfile[6] == 32'd0)  &&
            (uut.RegFiles.Regfile[7] == 32'd11)) begin
            $display("==========================================================");
            $display("PASS: All multi-cycle execution, FSM and branch tests passed.");
            $display("==========================================================");
        end else begin
            $display("==========================================================");
            $display("FAIL: Data mismatch detected in Multi-Cycle Core!");
            $display("x1=%0d x2=%0d x3=%0d mem[1]=%0d x4=%0d x5=%0d x6=%0d x7=%0d",
                     uut.RegFiles.Regfile[1], uut.RegFiles.Regfile[2],
                     uut.RegFiles.Regfile[3], uut.DMEM.RAM[1],
                     uut.RegFiles.Regfile[4], uut.RegFiles.Regfile[5],
                     uut.RegFiles.Regfile[6], uut.RegFiles.Regfile[7]);
            $display("==========================================================");
        end
        $finish;
    end
endmodule
